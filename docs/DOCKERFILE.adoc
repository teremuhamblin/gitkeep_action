2. 🐳 Dockerfile — Version Fusion Masterclass

[source,docker]
----
FROM python:3.11-slim

WORKDIR /app
COPY . .

RUN pip install --no-cache-dir -r requirements.txt || true

CMD ["python3", "src/main.py"]
----
